    """Extract every .zip under root (except ('stage1_train.zip',)), then delete it."""
    for z in _glob.glob(os.path.join(root, "*.zip")):
        if os.path.basename(z) in ('stage1_train.zip',):

        with zipfile.ZipFile(z) as f:
            f.extractall(root)
        os.remove(z)
